# Git command lines nv's `git` does not speak. Each must fail under the shim
# and say it is not supported, rather than do something else. One per row,
# run in a repository with two commits on `main` and a branch `side`.

rebase side
cherry-pick side
revert HEAD

blame a.txt
clean -fd

for-each-ref

config --global user.name someone
config --unset user.name
ls-tree -l HEAD
cat-file -s HEAD
cat-file --batch
rev-parse --git-dir
rev-parse --verify HEAD
init --bare elsewhere
fsck --lost-found

# Options that are spoken, with arguments that are not.

config core.editor vim

# Paths given to show and diff are names, a file or a directory: Git's patterns and magic are not.
show HEAD -- '*.txt'
diff HEAD~1 HEAD -- ':(exclude)a.txt'
show HEAD -- :/a.txt
